## src/pages/Login.tsx
import React, { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { useAuth } from '../contexts/AuthContext';
import { TextField, Button, Paper, Grid, Typography } from '@mui/material';

export const Login: React.FC = () => {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const { signIn } = useAuth();
  const navigate = useNavigate();

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    try {
      await signIn(email, password);
      navigate('/');
    } catch (error) {
      console.error('Erro ao fazer login:', error);
    }
  };

  return (
    <Grid container justifyContent="center" alignItems="center" style={{ minHeight: '100vh' }}>
      <Grid item xs={12} sm={8} md={6} lg={4}>
        <Paper elevation={3} sx={{ p: 4 }}>
          <Typography variant="h4" align="center" gutterBottom>
            Login
          </Typography>
          <form onSubmit={handleSubmit}>
            <TextField
              fullWidth
              label="Email"
              type="email"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              margin="normal"
              required
            />
            <TextField
              fullWidth
              label="Senha"
              type="password"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              margin="normal"
              required
            />
            <Button
              type="submit"
              variant="contained"
              color="primary"
              fullWidth
              sx={{ mt: 2 }}
            >
              Entrar
            </Button>
          </form>
        </Paper>
      </Grid>
    </Grid>
  );
};

## src/pages/Dashboard.tsx
import React from 'react';
import { useCards, useTransactions } from '../hooks';
import { Card, CardContent, Grid, Typography } from '@mui/material';
import { BarChart, Bar, XAxis, YAxis, CartesianGrid, Tooltip, Legend } from 'recharts';

export const Dashboard: React.FC = () => {
  const { cards, isLoading: cardsLoading } = useCards();
  const { transactions, isLoading: transactionsLoading } = useTransactions();

  if (cardsLoading || transactionsLoading) {
    return <div>Carregando...</div>;
  }

  const totalBalance = cards?.reduce((sum, card) => sum + card.balance, 0) || 0;
  const totalLimit = cards?.reduce((sum, card) => sum + card.limit, 0) || 0;
  const availableLimit = totalLimit - totalBalance;

  const monthlyExpenses = transactions
    ?.filter(t => t.type === 'expense')
    .reduce((sum, t) => sum + t.amount, 0) || 0;

  const monthlyIncome = transactions
    ?.filter(t => t.type === 'income')
    .reduce((sum, t) => sum + t.amount, 0) || 0;

  const monthlyBalance = monthlyIncome - monthlyExpenses;

  return (
    <Grid container spacing={3}>
      <Grid item xs={12} sm={6} md={3}>
        <Card>
          <CardContent>
            <Typography color="textSecondary" gutterBottom>
              Saldo Total
            </Typography>
            <Typography variant="h5">
              R$ {totalBalance.toFixed(2)}
            </Typography>
          </CardContent>
        </Card>
      </Grid>
      <Grid item xs={12} sm={6} md={3}>
        <Card>
          <CardContent>
            <Typography color="textSecondary" gutterBottom>
              Limite Total
            </Typography>
            <Typography variant="h5">
              R$ {totalLimit.toFixed(2)}
            </Typography>
          </CardContent>
        </Card>
      </Grid>
      <Grid item xs={12} sm={6} md={3}>
        <Card>
          <CardContent>
            <Typography color="textSecondary" gutterBottom>
              Limite Disponível
            </Typography>
            <Typography variant="h5">
              R$ {availableLimit.toFixed(2)}
            </Typography>
          </CardContent>
        </Card>
      </Grid>
      <Grid item xs={12} sm={6} md={3}>
        <Card>
          <CardContent>
            <Typography color="textSecondary" gutterBottom>
              Saldo Mensal
            </Typography>
            <Typography variant="h5">
              R$ {monthlyBalance.toFixed(2)}
            </Typography>
          </CardContent>
        </Card>
      </Grid>
      
      <Grid item xs={12}>
        <Card>
          <CardContent>
            <Typography variant="h6" gutterBottom>
              Despesas Mensais
            </Typography>
            <BarChart
              width={500}
              height={300}
              data={transactions?.filter(t => t.type === 'expense')}
            >
              <CartesianGrid strokeDasharray="3 3" />
              <XAxis dataKey="date" />
              <YAxis />
              <Tooltip />
              <Legend />
              <Bar dataKey="amount" fill="#8884d8" />
            </BarChart>
          </CardContent>
        </Card>
      </Grid>
    </Grid>
  );
};

## src/components/PrivateRoute.tsx
import React from 'react';
import { Navigate } from 'react-router-dom';
import { useAuth } from '../contexts/AuthContext';

interface PrivateRouteProps {
  children: React.ReactNode;
  path: string;
}

export const PrivateRoute: React.FC<PrivateRouteProps> = ({ children, path }) => {
  const { user, loading } = useAuth();

  if (loading) {
    return <div>Carregando...</div>;
  }

  if (!user) {
    return <Navigate to="/login" state={{ from: path }} replace />;
  }

  return <>{children}</>;
};

## src/components/Subscription.tsx
import React, { useState } from 'react';
import { useAuth } from '../contexts/AuthContext';
import { Button, Card, CardContent, Typography, Grid } from '@mui/material';
import { createSubscription } from '../lib/mercadopago';

export const Subscription: React.FC = () => {
  const { user } = useAuth();
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');

  const handleSubscribe = async () => {
    try {
      setLoading(true);
      setError('');
      
      const preference = await createSubscription(user?.email || '');
      window.location.href = preference.init_point;
    } catch (err) {
      setError('Erro ao criar assinatura. Tente novamente mais tarde.');
      console.error(err);
    } finally {
      setLoading(false);
    }
  };

  return (
    <Grid container justifyContent="center" alignItems="center" style={{ minHeight: '100vh' }}>
      <Grid item xs={12} sm={8} md={6}>
        <Card>
          <CardContent>
            <Typography variant="h4" align="center" gutterBottom>
              Plano Premium
            </Typography>
            <Typography variant="h6" align="center" gutterBottom>
              R$ 18,90/mês
            </Typography>
            <Typography variant="body1" paragraph>
              Benefícios do plano premium:
            </Typography>
            <ul>
              <li>Análise avançada de gastos</li>
              <li>Relatórios personalizados</li>
              <li>Suporte prioritário</li>
              <li>Recursos exclusivos</li>
            </ul>
            <Button
              variant="contained"
              color="primary"
              fullWidth
              onClick={handleSubscribe}
              disabled={loading}
            >
              {loading ? 'Processando...' : 'Assinar Agora'}
            </Button>
            {error && (
              <Typography color="error" align="center" sx={{ mt: 2 }}>

              </Typography>
            )}
          </CardContent>
        </Card>
      </Grid>
    </Grid>
  );
};
