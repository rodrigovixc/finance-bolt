# Especificação do Projeto Bolt Finance

## Visão Geral
Sistema de controle financeiro pessoal com autenticação, gerenciamento de cartões, transações, categorias e tipos de renda.

## Funcionalidades Principais

### 1. Autenticação e Autorização
- Registro de usuários com email e senha
- Login com email e senha
- Recuperação de senha
- Proteção de rotas autenticadas
- Gerenciamento de sessão
- Logout

### 2. Gerenciamento de Cartões
- Cadastro de cartões de crédito
  - Nome do cartão
  - Limite disponível
  - Saldo atual
  - Data de vencimento
- Listagem de cartões
- Edição de cartões
- Exclusão de cartões
- Visualização de saldo e limite disponível

### 3. Gerenciamento de Transações
- Cadastro de transações
  - Descrição
  - Valor
  - Data
  - Tipo (receita/despesa)
  - Cartão associado
  - Categoria
  - Tipo de renda (para receitas)
  - Recorrência (opcional)
  - Parcelamento (opcional)
- Listagem de transações
- Filtros por:
  - Data
  - Tipo
  - Categoria
  - Cartão
- Edição de transações
- Exclusão de transações

### 4. Gerenciamento de Categorias
- Cadastro de categorias
  - Nome
  - Descrição
  - Cor
  - Ícone
- Listagem de categorias
- Edição de categorias
- Exclusão de categorias

### 5. Gerenciamento de Tipos de Renda
- Cadastro de tipos de renda
  - Nome
  - Descrição
- Listagem de tipos de renda
- Edição de tipos de renda
- Exclusão de tipos de renda

### 6. Dashboard
- Visão geral do saldo
- Gráfico de despesas por categoria
- Gráfico de despesas mensais
- Total de receitas do mês
- Total de despesas do mês
- Saldo mensal
- Limite total dos cartões
- Limite disponível

### 7. Assinatura Premium
- Integração com Mercado Pago
- Plano mensal de R$ 18,90
- Página de assinatura
- Status de pagamento
- Webhook para notificações
- Gerenciamento de status da assinatura

```sql
    created_at TIMESTAMP,
    updated_at TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id)

CREATE TABLE transactions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED,
    description VARCHAR(255),

    date DATE,
    type ENUM('income', 'expense'),
    card_id BIGINT UNSIGNED NULL,
    category_id BIGINT UNSIGNED NULL,
    income_type_id BIGINT UNSIGNED NULL,
    is_recurring BOOLEAN DEFAULT FALSE,
    recurrence_period VARCHAR(50) NULL,
    recurrence_end_date DATE NULL,
    installments JSON NULL,
    created_at TIMESTAMP,
    updated_at TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (card_id) REFERENCES cards(id),
    FOREIGN KEY (category_id) REFERENCES categories(id),
    FOREIGN KEY (income_type_id) REFERENCES income_types(id)
);

-- Tabela de assinaturas
CREATE TABLE subscriptions (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED,

    plan ENUM('free', 'premium'),
    start_date TIMESTAMP,
    end_date TIMESTAMP,
    created_at TIMESTAMP,
    updated_at TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id)
```

5. Controladores
```php
// app/Http/Controllers/AuthController.php
namespace App\Http\Controllers;

use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Auth;

class AuthController extends Controller
{
    public function register(Request $request)
    {
        $request->validate([
            'name' => 'required|string|max:255',
            'email' => 'required|string|email|max:255|unique:users',
            'password' => 'required|string|min:8',
        ]);

        $user = User::create([
            'name' => $request->name,
            'email' => $request->email,
            'password' => Hash::make($request->password),
        ]);

        $token = $user->createToken('auth_token')->plainTextToken;

        return response()->json([
            'user' => $user,
            'token' => $token,
        ]);
    }

    public function login(Request $request)
    {
        if (!Auth::attempt($request->only('email', 'password'))) {
            return response()->json([
                'message' => 'Credenciais inválidas'
            ], 401);
        }

        $user = User::where('email', $request->email)->firstOrFail();
        $token = $user->createToken('auth_token')->plainTextToken;

        return response()->json([
            'user' => $user,
            'token' => $token,
        ]);
    }

    public function logout(Request $request)
    {
        $request->user()->currentAccessToken()->delete();
        return response()->json(['message' => 'Logout realizado com sucesso']);
    }
}

// app/Http/Controllers/CardController.php
namespace App\Http\Controllers;

use App\Models\Card;
use Illuminate\Http\Request;

class CardController extends Controller
{
    public function index()
    {
        return Card::where('user_id', auth()->id())->get();
    }

    public function store(Request $request)
    {
        $request->validate([
            'name' => 'required|string|max:255',
            'limit' => 'required|numeric|min:0',
            'balance' => 'required|numeric',
            'due_date' => 'required|date',
        ]);

        return Card::create([
            'user_id' => auth()->id(),
            'name' => $request->name,
            'limit' => $request->limit,
            'balance' => $request->balance,
            'due_date' => $request->due_date,
        ]);
    }

    public function update(Request $request, Card $card)
    {
        $this->authorize('update', $card);

        $request->validate([
            'name' => 'required|string|max:255',
            'limit' => 'required|numeric|min:0',
            'balance' => 'required|numeric',
            'due_date' => 'required|date',
        ]);

        $card->update($request->all());
        return $card;
    }

    public function destroy(Card $card)
    {
        $this->authorize('delete', $card);
        $card->delete();
        return response()->json(null, 204);
    }
}
```

```php
// app/Services/MercadoPagoService.php
namespace App\Services;

use MercadoPago\SDK;
use MercadoPago\Preference;

class MercadoPagoService
{
    public function __construct()
    {
        SDK::setAccessToken(config('services.mercadopago.access_token'));
    }

    public function createSubscription($user)
    {
        $preference = new Preference();
        
        $preference->items = [
            [
                'title' => 'Plano Premium - Bolt Finance',
                'unit_price' => 18.90,
                'quantity' => 1,
                'currency_id' => 'BRL',
            ]
        ];
        
        $preference->payer = [
            'email' => $user->email
        ];
        
        $preference->back_urls = [
            'success' => config('app.url') . '/payment/success',
            'failure' => config('app.url') . '/payment/failure',
            'pending' => config('app.url') . '/payment/pending'
        ];
        
        $preference->auto_return = 'approved';
        $preference->notification_url = config('app.url') . '/api/webhook/mercadopago';
        $preference->statement_descriptor = 'BOLT FINANCE';
        
        $preference->save();
        
        return $preference;
    }
}
```

```typescript
// src/contexts/AuthContext.tsx
import React, { createContext, useContext, useState, useEffect } from 'react';
import api from '../services/api';

interface AuthContextData {
  user: User | null;
  loading: boolean;
  signIn: (email: string, password: string) => Promise<void>;
  signUp: (name: string, email: string, password: string) => Promise<void>;
  signOut: () => void;
}

const AuthContext = createContext<AuthContextData>({} as AuthContextData);

export const AuthProvider: React.FC = ({ children }) => {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const token = localStorage.getItem('token');
    if (token) {
      api.defaults.headers.authorization = `Bearer ${token}`;
      // Carregar dados do usuário
    }
    setLoading(false);
  }, []);

  const signIn = async (email: string, password: string) => {
    const response = await api.post('/login', { email, password });
    const { user, token } = response.data;
    
    localStorage.setItem('token', token);
    api.defaults.headers.authorization = `Bearer ${token}`;
    setUser(user);
  };

  const signUp = async (name: string, email: string, password: string) => {
    const response = await api.post('/register', { name, email, password });
    const { user, token } = response.data;
    
    localStorage.setItem('token', token);
    api.defaults.headers.authorization = `Bearer ${token}`;
    setUser(user);
  };

  const signOut = () => {
    localStorage.removeItem('token');
    api.defaults.headers.authorization = undefined;
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ user, loading, signIn, signUp, signOut }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => useContext(AuthContext);
```

```typescript
// src/hooks/useCards.ts
import { useQuery, useMutation, useQueryClient } from 'react-query';
import api from '../services/api';

export const useCards = () => {
  const queryClient = useQueryClient();

  const { data: cards, isLoading } = useQuery('cards', async () => {
    const response = await api.get('/cards');
    return response.data;
  });

  const createCard = useMutation(
    (newCard: Card) => api.post('/cards', newCard),
    {
      onSuccess: () => {
        queryClient.invalidateQueries('cards');
      },
    }
  );

  const updateCard = useMutation(
    ({ id, ...data }: Card) => api.put(`/cards/${id}`, data),
    {
      onSuccess: () => {
        queryClient.invalidateQueries('cards');
      },
    }
  );

  const deleteCard = useMutation(
    (id: string) => api.delete(`/cards/${id}`),
    {
      onSuccess: () => {
        queryClient.invalidateQueries('cards');
      },
    }
  );

  return {
    cards,
    isLoading,
    createCard,
    updateCard,
    deleteCard,
  };
};
```

```typescript
import { useCards, useTransactions } from '../hooks';

    return <div>Carregando...</div>;
  }

  const totalBalance = cards?.reduce((sum, card) => sum + card.balance, 0) || 0;
```
